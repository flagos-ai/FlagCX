// ===========================================================================
// S20 minimal probe (diagnostic only). Append to the END of
// test/kernel/musa/device_ir.cu. The public S16-S25 kernels are not modified.
//
// One kernel, mode selects the step. Every mode runs in its own MPI job so an
// illegal access cannot hide the others.
//
//   7  = ResetSignal only           (user slots 0..3, one thread per block)
//   8  = ReadSignal only            (same slots)
//   9  = INTER SignalInc + WaitSignal (this block's slot)
//   10 = Reset -> INTER barrier -> SignalInc -> Wait -> Read (S20 combo 0)
//   11 = K11-shaped control: context 0, slot 0, BLOCK coop, WORLD barrier
//        (must pass; otherwise the probe harness itself is wrong)
// ===========================================================================
__global__ void musaS20ProbeKernel(const void *opaque, int *out, int mode) {
  const flagcxDevComm *comm = static_cast<const flagcxDevComm *>(opaque);
  const int nRanks = comm->getSize();
  const int intraSize = comm->getIntraSize();
  const int nNodes = intraSize > 0 ? nRanks / intraSize : 1;
  const int nodeIdx = intraSize > 0 ? comm->getRank() / intraSize : 0;
  const int peer = (nodeIdx + 1) % nNodes;
  const int nContexts = comm->getContextCount();
  const int tid = (int)FLAGCX_THREAD_IDX_X;
  const int blk = (int)FLAGCX_BLOCK_IDX_X;
  // The control uses one context and one slot, exactly like K11.
  const int ctx = (mode == 11)
                      ? 0
                      : (nContexts > 0 ? blk % nContexts : 0);
  const flagcxDevSignal_t slot =
      (flagcxDevSignal_t)((mode == 11) ? 0 : blk);

  if (mode == 7) {
    if (tid == 0) {
      flagcxDevResetSignal(opaque, ctx, slot);
      out[blk * 16] = 1;
    }
    return;
  }
  if (mode == 8) {
    if (tid == 0) {
      uint64_t value = flagcxDevReadSignal(opaque, slot, kCompletionBits, ctx,
                                           flagcxDeviceMemoryOrderAcquire);
      out[blk * 16] = 1;
      out[blk * 16 + 1] = (int)value;
    }
    return;
  }
  if (mode == 9) {
    if (tid == 0) {
      flagcxDevSignalInc(opaque, FLAGCX_TEAM_INTER, peer, slot, ctx,
                         FLAGCX_COOP_THREAD, flagcxDeviceScopeSystem);
      flagcxDevWaitSignal(opaque, slot, 1, kCompletionBits, ctx,
                          FLAGCX_COOP_THREAD, flagcxDeviceMemoryOrderAcquire);
      out[blk * 16] = 1;
    }
    return;
  }

  // modes 10 and 11: full reset -> barrier -> signal -> wait -> read sequence.
  if (tid == 0) {
    flagcxDevResetSignal(opaque, ctx, slot);
    (void)flagcxDevReadSignal(opaque, slot, kCompletionBits, ctx,
                              flagcxDeviceMemoryOrderAcquire);
  }
  FLAGCX_DEVICE_SYNC_THREADS();
  flagcxDevBarrierSync(opaque,
                       (mode == 11) ? FLAGCX_TEAM_WORLD : FLAGCX_TEAM_INTER,
                       (uint32_t)blk, ctx, FLAGCX_COOP_BLOCK,
                       flagcxDeviceMemoryOrderAcqRel, flagcxDeviceScopeSystem);
  if (tid == 0) {
    flagcxDevSignalInc(opaque, FLAGCX_TEAM_INTER, peer, slot, ctx,
                       FLAGCX_COOP_THREAD, flagcxDeviceScopeSystem);
  }
  if (mode == 10) {
    if (tid == 0) {
      flagcxDevWaitSignal(opaque, slot, 1, kCompletionBits, ctx,
                          FLAGCX_COOP_THREAD, flagcxDeviceMemoryOrderAcquire);
      (void)flagcxDevReadSignal(opaque, slot, kCompletionBits, ctx,
                                flagcxDeviceMemoryOrderAcquire);
      out[blk * 16] = 1;
    }
    return;
  }
  // A block-wide coop must be entered by every thread of the CTA.
  flagcxDevWaitSignal(opaque, slot, 1, kCompletionBits, ctx, FLAGCX_COOP_BLOCK,
                      flagcxDeviceMemoryOrderAcquire);
  FLAGCX_DEVICE_SYNC_THREADS();
  if (tid == 0)
    out[blk * 16] = 1;
}

extern "C" flagcxResult_t flagcxMusaS20ProbeRun(int mode, const void *opaque,
                                                int *out,
                                                flagcxStream_t stream) {
  if (!opaque || !out || !stream || mode < 7 || mode > 11)
    return flagcxInvalidArgument;
  musaError_t prior = musaGetLastError();
  std::fprintf(stderr, "S20 PROBE mode=%d prior=%d (%s)\n", mode, int(prior),
               musaGetErrorString(prior));
  std::fflush(stderr);
  if (prior != musaSuccess)
    return flagcxUnhandledDeviceError;
  musaS20ProbeKernel<<<4, 128, 0, stream->base>>>(opaque, out, mode);
  musaError_t launch = musaGetLastError();
  std::fprintf(stderr, "S20 PROBE mode=%d launch=%d (%s)\n", mode, int(launch),
               musaGetErrorString(launch));
  std::fflush(stderr);
  if (launch != musaSuccess)
    return flagcxUnhandledDeviceError;
  musaError_t sync = musaStreamSynchronize(stream->base);
  std::fprintf(stderr, "S20 PROBE mode=%d sync=%d (%s)\n", mode, int(sync),
               musaGetErrorString(sync));
  std::fflush(stderr);
  if (sync != musaSuccess)
    return flagcxUnhandledDeviceError;
  return flagcxSuccess;
}
